  async function streamReply(userText){
    var L=window.__nsp_L||EN;
    history.push({role:"user",content:userText});
    updateHumanBtn();
    appendMsg("user",userText);
    appendTyping();
    streaming=true;sendBtn.disabled=true;
    var bub=null,full="";
    try{
      var resp=await fetch("/api/support/chat",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({sid:getSid(),messages:history.slice(-20),lang:window.__nsp_lang||"en",page:location.pathname})});
      if(!resp.ok||!resp.body){removeTyping();appendMsg("sys",L.net_err);streaming=false;sendBtn.disabled=false;return;}
      var reader=resp.body.getReader(),decoder=new TextDecoder("utf-8"),buf="";
      while(true){
        var r=await reader.read();
        if(r.done)break;
        buf+=decoder.decode(r.value,{stream:true});
        var lines=buf.split("\n\n");buf=lines.pop()||"";
        for(var i=0;i<lines.length;i++){
          var lne=lines[i];
          if(lne.indexOf("event: escalate_available")===0){escalateOk=true;updateHumanBtn();continue;}
          if(lne.indexOf("event: escalated")===0){
            var dl=lne.split("\n").find(function(x){return x.indexOf("data:")===0;});
            if(dl){try{var ed=JSON.parse(dl.slice(5).trim());appendMsg("sys",L.human_done+" #"+ed.ticket_id);}catch(e){}}
            continue;
          }
          if(lne.indexOf("data:")!==0)continue;
          var data=lne.replace(/^data:\s*/,"").trim();
          if(data==="[DONE]")continue;
          try{
            var obj=JSON.parse(data);
            var choices=obj.choices||[];
            var delta=(choices[0]||{}).delta||{};
            var c=delta.content||"";
            if(c){
              if(!bub){removeTyping();bub=appendMsg("bot","");}
              full+=c;bub.textContent=full;
              msgsEl.scrollTop=msgsEl.scrollHeight;
            }
          }catch(e){}
        }
      }
      if(!bub){removeTyping();}
      if(full)history.push({role:"assistant",content:full});
    }catch(e){removeTyping();appendMsg("sys",L.net_err);}
    streaming=false;sendBtn.disabled=false;
  }
